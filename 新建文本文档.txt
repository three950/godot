## 最佳实践：简洁高效的迭代版本。线性堆叠：单链表 + 数组
func collect_cards(root_card: Control) -> Array:
	var result: Array = []
	var queue: Array = [root_card]
	
	while not queue.is_empty():
		var card = queue.pop_front()
		result.append(card)
		
		if "stacked_cards" in card:
			var stacked = card.stacked_cards
			if stacked != null:
				for child in stacked:
					queue.append(child)
	
	return result
